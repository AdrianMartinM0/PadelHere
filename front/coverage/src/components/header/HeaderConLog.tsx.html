const navUser = [

    { to: "/app/chats", label: "Chats" },
    { to: "/app/clubs", label: "Clubes" },
    { to: "/app/mis-partidos", label: "Mis Partidos" },
    { to: "/app/reservas", label: "Mis Reservas" },
];
 
const HeaderConLog = () => {

    return (

                "
                            >

                        </Link>
                    ))}

            {userType === "club" && (
